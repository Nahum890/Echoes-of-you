### 11. EXAMPLES

#### Example 11.1: Automated Assertion Method in C#
```csharp
public static bool ValidateLightCount(out string error)
{
    Light[] lights = Object.FindObjectsOfType<Light>();
    if (lights.Length > 48)
    {
        error = $"FAIL-LGT-01: Light count ({lights.Length}) exceeds 48 cap.";
        return false;
    }
    error = null;
    return true;
}

// VAL-C-02: Hard Shadows enforcement
public static bool ValidateShadowType(out string error)
{
    foreach (Light light in Object.FindObjectsOfType<Light>())
    {
        if (light.shadows != LightShadows.Hard)
        {
            error = $"FAIL-LGT-02: Light '{light.name}' uses {light.shadows}, must be LightShadows.Hard.";
            return false;
        }
    }
    error = null;
    return true;
}
```

#### Example 11.2: Echo Button Test — Deterministic Solvability Algorithm (VAL-B-01)
```csharp
// Echo Button Test: BFS graph traversal to confirm puzzle requires Echo
// Spec: [VAL-B-01] — Puzzle requires Echo recording or dual-body presence
public static bool EchoButtonTest(LevelBlueprint blueprint, out string error)
{
    // Phase 1: Build signal graph from blueprint puzzle components
    var signalGraph = new Dictionary<string, List<string>>();
    var echoOnlyNodes = new HashSet<string>();

    foreach (var comp in blueprint.puzzle.components)
    {
        signalGraph[comp.id] = comp.signalOutputs ?? new List<string>();
        if (comp.requiresEcho || comp.type == "PressurePlate_EchoOnly")
            echoOnlyNodes.Add(comp.id);
    }

    // Phase 2: BFS from all LevelExit nodes backwards
    // Check if any path from root requires an Echo-only node
    bool requiresEcho = false;
    var visited = new HashSet<string>();
    var queue = new Queue<string>();
    foreach (var exit in blueprint.puzzle.exitNodes) queue.Enqueue(exit);

    while (queue.Count > 0)
    {
        string node = queue.Dequeue();
        if (!visited.Add(node)) continue;
        if (echoOnlyNodes.Contains(node)) { requiresEcho = true; break; }
        foreach (var dep in signalGraph.GetValueOrDefault(node, new List<string>()))
            queue.Enqueue(dep);
    }

    if (!requiresEcho)
    {
        error = "FAIL-PUZ-01: No Echo-required node found in puzzle graph. Puzzle is solvable without Echo recording.";
        return false;
    }
    error = null;
    return true;
}
```
